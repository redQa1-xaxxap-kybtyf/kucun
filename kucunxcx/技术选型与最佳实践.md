### 1. 网络请求封装 (基于Vant组件)

```typescript
// utils/request.ts
import { StorageService } from './storage';

interface RequestOptions {
  url: string;
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE';
  data?: any;
  header?: any;
  showLoading?: boolean;
  showError?: boolean;
}

interface ApiResponse<T = any> {
  success: boolean;
  data?: T;
  error?: string;
  message?: string;
}

const API_BASE_URL = 'https://your-domain.com/api';
const REQUEST_TIMEOUT = 30000;

export class RequestService {
  /**
   * 统一请求方法
   */
  static async request<T>(options: RequestOptions): Promise<ApiResponse<T>> {
    const {
      url,
      method = 'GET',
      data,
      header = {},
      showLoading = true,
      showError = true,
    } = options;

    // 显示加载提示
    if (showLoading) {
      wx.showLoading({ title: '加载中...', mask: true });
    }

    try {
      // 获取token
      const token = StorageService.getToken();

      // 构建请求头
      const requestHeader = {
        'Content-Type': 'application/json',
        ...(token && { Authorization: `Bearer ${token}` }),
        ...header,
      };

      // 发起请求
      const res = await wx.request({
        url: `${API_BASE_URL}${url}`,
        method,
        data,
        header: requestHeader,
        timeout: REQUEST_TIMEOUT,
      });

      // 隐藏加载提示
      if (showLoading) {
        wx.hideLoading();
      }

      // 处理响应
      if (res.statusCode === 200) {
        const response = res.data as ApiResponse<T>;

        if (response.success) {
          return response;
        } else {
          // 业务错误
          if (showError) {
            this.showError(response.error || '请求失败');
          }
          return response;
        }
      } else if (res.statusCode === 401) {
        // Token过期
        this.handleUnauthorized();
        throw new Error('未授权,请重新登录');
      } else {
        // HTTP错误
        const errorMsg = `请求失败 (${res.statusCode})`;
        if (showError) {
          this.showError(errorMsg);
        }
        throw new Error(errorMsg);
      }
    } catch (error) {
      // 隐藏加载提示
      if (showLoading) {
        wx.hideLoading();
      }

      // 处理异常
      const errorMsg = error instanceof Error ? error.message : '网络请求失败';
      if (showError) {
        this.showError(errorMsg);
      }

      throw error;
    }
  }

  /**
   * GET请求
   */
  static get<T>(url: string, params?: any, options?: Partial<RequestOptions>) {
    const query = params ? '?' + new URLSearchParams(params).toString() : '';
    return this.request<T>({ url: url + query, method: 'GET', ...options });
  }

  /**
   * POST请求
   */
  static post<T>(url: string, data?: any, options?: Partial<RequestOptions>) {
    return this.request<T>({ url, method: 'POST', data, ...options });
  }

  /**
   * PUT请求
   */
  static put<T>(url: string, data?: any, options?: Partial<RequestOptions>) {
    return this.request<T>({ url, method: 'PUT', data, ...options });
  }

  /**
   * DELETE请求
   */
  static delete<T>(url: string, options?: Partial<RequestOptions>) {
    return this.request<T>({ url, method: 'DELETE', ...options });
  }

  /**
   * 显示错误提示 (使用Vant Toast)
   */
  private static showError(message: string) {
    // 使用Vant Toast组件
    wx.showToast({
      title: message,
      icon: 'none',
      duration: 2000,
    });
  }

  /**
   * 处理未授权
   */
  private static handleUnauthorized() {
    StorageService.clearAuth();
    wx.showToast({
      title: '登录已过期',
      icon: 'none',
      duration: 2000,
      success: () => {
        setTimeout(() => {
          wx.navigateTo({ url: '/pages/auth/login/login' });
        }, 2000);
      },
    });
  }
}
```

### 2. 产品列表页 (使用Vant组件)

```typescript
// pages/products/list/list.ts
import { ProductService } from '../../../services/product';
import type { Product } from '../../../types/product';

Page({
  data: {
    products: [] as Product[],
    loading: false,
    finished: false,
    page: 1,
    limit: 20,
    searchValue: '',

    // Vant组件状态
    pullDownRefreshing: false,
    loadingMore: false,
  },

  onLoad() {
    this.loadProducts();
  },

  /**
   * 加载产品列表
   */
  async loadProducts(refresh = false) {
    if (this.data.loading) return;

    const page = refresh ? 1 : this.data.page;

    this.setData({ loading: true });

    try {
      const response = await ProductService.getProducts({
        page,
        limit: this.data.limit,
        search: this.data.searchValue,
        includeInventory: true,
      });

      if (response.success && response.data) {
        const { products, pagination } = response.data;

        this.setData({
          products: refresh ? products : [...this.data.products, ...products],
          page: page + 1,
          finished: products.length < this.data.limit,
          loading: false,
          pullDownRefreshing: false,
        });
      }
    } catch (error) {
      this.setData({
        loading: false,
        pullDownRefreshing: false,
      });
    }
  },

  /**
   * 下拉刷新
   */
  onPullDownRefresh() {
    this.setData({ pullDownRefreshing: true });
    this.loadProducts(true);
  },

  /**
   * 上拉加载更多
   */
  onReachBottom() {
    if (!this.data.finished && !this.data.loading) {
      this.loadProducts();
    }
  },

  /**
   * 搜索
   */
  onSearch(event: any) {
    this.setData({
      searchValue: event.detail,
      page: 1,
    });
    this.loadProducts(true);
  },

  /**
   * 跳转到产品详情
   */
  onProductTap(event: any) {
    const { id } = event.currentTarget.dataset;
    wx.navigateTo({
      url: `/pages/products/detail/detail?id=${id}`,
    });
  },
});
```

## 📊 成本效益分析

### 原生 + Vant Weapp 方案

#### 成本

```yaml
学习成本: ⭐⭐ (低)
  - 微信小程序基础: 2-3天
  - Vant Weapp: 1天
  - TypeScript: 已掌握

开发成本: ⭐⭐ (低)
  - 初期搭建: 1天
  - 核心功能: 3-4天
  - 用户系统: 2天
  - 优化测试: 2天
  总计: 8-10天

维护成本: ⭐ (极低)
  - 原生API稳定
  - Vant Weapp持续更新
  - 无跨端适配成本
```

#### 收益

```yaml
性能: ⭐⭐⭐⭐⭐
  - 首屏 <1秒
  - 60fps流畅
  - 包体积最小

开发效率: ⭐⭐⭐⭐
  - Vant组件开箱即用
  - TypeScript类型安全
  - 调试体验优秀

长期维护: ⭐⭐⭐⭐⭐
  - 代码简单清晰
  - 依赖少
  - 易于接手
```

### 跨端框架方案 (对比)

#### 额外成本

```yaml
学习成本: ⭐⭐⭐⭐ (高)
  - Taro/uni-app框架: 3-5天
  - 跨端概念: 2-3天
  - 平台差异: 持续学习

开发成本: ⭐⭐⭐ (中)
  - 框架配置: 1-2天
  - 平台适配: 持续投入
  - 调试复杂度: 高

  - 框架版本升级
  - 平台API变更适配

```

#### 收益不足

```yaml
对你的项目:
  - ❌ 无跨端需求 → 跨端能力浪费
```

## 📝 总结

### 为什么选择 **原生 + Vant Weapp**?

1. **✅ 符合项目需求**
   - 单端开发,无跨端诉求
   - 性能要求高
   - 开发周期紧

2. **✅ 技术成熟稳定**
   - 原生API完整
   - Vant生态成熟
   - 社区支持强

3. **✅ 开发效率高**
   - 学习曲线平缓
   - 组件开箱即用
   - 调试体验好

4. **✅ 长期维护友好**
   - 代码简单
   - 依赖少
   - 易于接手

### 不选择跨端框架的原因

1. **❌ 不符合项目需求**
   - 无H5/App需求
   - 跨端能力浪费
   - 增加复杂度

2. **❌ 性能有损失**
   - 首屏慢20-30%
   - 包体积大
   - 内存占用高

3. **❌ 学习成本高**
   - 框架学习周期长
   - 平台差异复杂
   - 调试困难

4. **❌ 维护成本高**
   - 框架升级风险
   - 平台适配持续
   - Bug排查复杂
