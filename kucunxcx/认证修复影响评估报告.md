
## ❌ 小程序认证失败的根本原因

### 问题链

```
小程序 → Bearer Token (存在请求头)
           ↓
      middleware.ts拦截
           ↓
      authMiddleware调用getToken()
           ↓
      getToken()查找NextAuth的session cookie
           ↓
      ❌ 找不到cookie（小程序用的是Bearer Token）
           ↓
      返回null
           ↓
      middleware返回401 (第131-143行)
```

### 核心问题

**getToken() 只识别NextAuth的JWT Cookie，不识别 Authorization: Bearer**

```typescript
// lib/auth-middleware.ts:122
const token = await getToken({
  req: request,
  secret: env.NEXTAUTH_SECRET,
});
// ❌ 这个函数只会查找 Cookie 中的 next-auth.session-token
// ❌ 不会处理 Authorization: Bearer <token>
```

**1. 创建 `lib/auth/bearer-token.ts`**:

```typescript
// 专门处理 Bearer Token 验证
import { prisma } from '@/lib/db';

export interface BearerTokenResult {
  success: boolean;
  user?: {
    id: string;
    email: string;
    name: string;
    username: string;
    role: string;
    status: string;
  };
}

// ⚡ 使用内存缓存减少数据库查询
const tokenCache = new Map<
  string,
  {
    user: any;
    expiresAt: number;
  }
>();

const CACHE_TTL = 5 * 60 * 1000; // 5分钟缓存

export async function verifyBearerToken(
  token: string
): Promise<BearerTokenResult> {
  // 检查缓存
  const cached = tokenCache.get(token);
  if (cached && Date.now() < cached.expiresAt) {
    return { success: true, user: cached.user };
  }

  try {
    const session = await prisma.session.findUnique({
      where: { sessionToken: token },
      include: {
        user: {
          select: {
            id: true,
            email: true,
            name: true,
            username: true,
            role: true,
            status: true,
          },
        },
      },
    });

    if (!session || session.expires < new Date()) {
      tokenCache.delete(token);
      return { success: false };
    }

    // 更新缓存
    tokenCache.set(token, {
      user: session.user,
      expiresAt: Date.now() + CACHE_TTL,
    });

    return { success: true, user: session.user };
  } catch (error) {
    return { success: false };
  }
}
```

```typescript
// 在文件顶部添加import
import { verifyBearerToken } from '@/lib/auth/bearer-token';

// 在 authMiddleware 函数中（第120行之后）
export async function authMiddleware(request: NextRequest) {
  // ... 前面代码保持不变 ...

  try {
    let token: any = null;
    let source: 'nextauth' | 'bearer' = 'nextauth';

    // 1. 尝试 Bearer Token (小程序)
    const authHeader = request.headers.get('Authorization');
    if (authHeader?.startsWith('Bearer ')) {
      const bearerToken = authHeader.substring(7);
      const result = await verifyBearerToken(bearerToken);

      if (result.success && result.user) {
        token = result.user;
        token.sub = result.user.id; // 兼容 NextAuth 格式
        source = 'bearer';
      }
    }

    // 2. 如果Bearer Token失败，尝试NextAuth (Web前端)
    if (!token) {
      token = await getToken({
        req: request,
        secret: env.NEXTAUTH_SECRET,
      });
    }

    // ... 后续代码保持完全不变 ...
    // 第130-210行的逻辑完全不需要修改
  }
}
```
