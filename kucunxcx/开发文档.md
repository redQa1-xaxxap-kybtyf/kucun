# 库存管理小程序开发文档

## 📋 项目概述

### 项目背景

本小程序是瓷砖行业库存管理系统的移动端展示应用,主要面向游客和注册用户提供产品浏览和库存查询功能。

### 核心功能

1. **产品展示** - 产品列表、详情、图片浏览
2. **库存查询** - 实时库存信息、批次查询
3. **用户系统** - 登录注册、权限管理
4. **分类浏览** - 按产品分类筛选

### 用户角色

- **游客** - 可浏览产品和查看库存信息
- **注册用户** - 完整功能访问权限

```typescript
const PERMISSIONS = {

  user: ['products:view', 'inventory:view', 'inventory:export'],

    'products:view',

    'orders:create',
  ],
  manager: ['*'],
  admin: ['*'],
};
```

### 路由守卫

```typescript
// utils/auth-guard.ts
export function requireAuth(
  userRole: string,
  requiredPermission: string
): boolean {
  const userPermissions = PERMISSIONS[userRole] || [];
  return (
    userPermissions.includes('*') ||
    userPermissions.includes(requiredPermission)
  );
}
```

### 网络请求封装

```typescript
// utils/request.ts
interface RequestOptions {
  url: string;
  method: 'GET' | 'POST' | 'PUT' | 'DELETE';
  data?: any;
  header?: any;
  needAuth?: boolean;
}

export function request<T>(options: RequestOptions): Promise<T> {
  const { url, method, data, header = {}, needAuth = true } = options;

  // 添加认证token
  if (needAuth) {
    const token = getToken();
    if (token) {
      header['Authorization'] = `Bearer ${token}`;
    }
  }

  return new Promise((resolve, reject) => {
    wx.request({
      url: `${API_BASE_URL}${url}`,
      method,
      data,
      header: {
        ...DEFAULT_HEADERS,
        ...header,
      },
      success: res => {
        if (res.statusCode === 200) {
          resolve(res.data as T);
        } else if (res.statusCode === 401) {
          // Token过期，跳转登录
          clearAuth();
          wx.navigateTo({ url: '/pages/auth/login/login' });
          reject(new Error('未授权'));
        } else {
          reject(new Error(res.data.error || '请求失败'));
        }
      },
      fail: reject,
    });
  });
}
```

### 1. 产品服务

```typescript
// services/product.service.ts
import { request } from '../utils/request';
import type { Product, ProductListResponse } from '../types/product';

export class ProductService {
  // 获取产品列表
  static async getProducts(params: {
    page?: number;
    limit?: number;
    search?: string;
    categoryId?: string;
  }): Promise<ProductListResponse> {
    const query = new URLSearchParams(params as any).toString();
    return request<ProductListResponse>({
      url: `/products?${query}`,
      method: 'GET',
    });
  }

  // 获取产品详情
  static async getProductDetail(id: string): Promise<Product> {
    return request<Product>({
      url: `/products/${id}`,
      method: 'GET',
    });
  }

  // 搜索产品
  static async searchProducts(keyword: string): Promise<Product[]> {
    return request<Product[]>({
      url: `/products?search=${keyword}`,
      method: 'GET',
    });
  }
}
```

### 3. 认证服务

```typescript
// services/auth.service.ts
import { request } from '../utils/request';
import { setUserInfo, setToken, clearAuth } from '../utils/storage';

export class AuthService {
  // 登录
  static async login(username: string, password: string) {
    const response = await request({
      url: '/auth/signin',
      method: 'POST',
      data: { username, password },
      needAuth: false,
    });

    if (response.success) {
      setUserInfo(response.data.user);
      setToken(response.data.token);
    }

    return response;
  }

  // 注册
  static async register(data: {
    username: string;
    email: string;
    password: string;
    name: string;
  }) {
    return request({
      url: '/auth/register',
      method: 'POST',
      data,
      needAuth: false,
    });
  }

  // 登出
  static logout() {
    clearAuth();
    wx.reLaunch({ url: '/pages/index/index' });
  }
}
```
