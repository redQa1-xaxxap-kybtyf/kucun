## 🔍 详细问题分析

### 问题1: 采购运费未分摊到库存成本 🔴

**严重程度**: 高  
**影响范围**: 成本核算、毛利计算、定价决策  
**业务影响**: 成本被低估，毛利被高估，可能导致定价过低

#### 当前状态

```typescript

  productId: string;
  quantity: number;
  unitPrice: number;

  });
}
```

**采购订单代码**（`app/actions/purchase-orders.ts`）:

```typescript
    totalAmount,
```

#### 问题根源

1. **采购订单的运费**记录在 `PurchaseOrder.expenseAmount`，但未分摊到 `PurchaseOrderItem`
2. **入库时**直接使用 `PurchaseOrderItem.unitPrice`，未加上分摊的运费
3. **库存成本**因此被低估

#### 影响示例

```
场景：采购100件瓷砖，单价10元/件，运费500元

当前错误做法：
- 入库成本：10元/件
- 库存总成本：1000元
- 销售50件，成本：500元
- 毛利：销售额 - 500元（错误，成本被低估）

正确做法：
- 分摊后成本：10 + 500/100 = 15元/件
- 库存总成本：1500元
- 销售50件，成本：750元
- 毛利：销售额 - 750元（正确）

差异：毛利被高估250元（50%）
```

### 方案1: 采购运费分摊到库存成本 🔴

**目标**: 确保采购运费正确分摊到每个产品的单位成本

#### 技术方案

**步骤1: 扩展PurchaseOrderItem表**（可选，推荐）

```prisma
model PurchaseOrderItem {
  // ... 现有字段
  allocatedExpense Float? @map("allocated_expense") // 分摊的运费
  unitCostWithExpense Float? @map("unit_cost_with_expense") // 含运费的单位成本
}
```

**步骤2: 采购订单确认时计算费用分摊**

```typescript
// app/actions/purchase-orders.ts

export async function confirmPurchaseOrder(orderId: string) {
  return await prisma.$transaction(async tx => {
    const order = await tx.purchaseOrder.findUnique({
      where: { id: orderId },
      include: { items: true },
    });

    // 1. 计算费用分摊
    const totalQuantity = order.items.reduce(
      (sum, item) => sum + item.quantity,
      0
    );
    const expensePerUnit = order.expenseAmount / totalQuantity;

    // 2. 更新每个明细的成本
    for (const item of order.items) {
      const allocatedExpense = expensePerUnit * item.quantity;
      const unitCostWithExpense = item.unitPrice + expensePerUnit;

      await tx.purchaseOrderItem.update({
        where: { id: item.id },
        data: {
          allocatedExpense,
          unitCostWithExpense,
        },
      });
    }

    // 3. 更新订单状态
    await tx.purchaseOrder.update({
      where: { id: orderId },
      data: { status: 'confirmed' },
    });
  });
}
```

**步骤3: 入库时使用含运费的成本**

```typescript
  // ...

```

**步骤1: 发货时计算费用分摊**

```typescript
// lib/api/handlers/sales-order-status.ts

import { allocateExpensesByValue } from '@/lib/services/factory-shipment-expense-service';

export async function shipSalesOrder(orderId: string) {
  return await prisma.$transaction(async tx => {
    const order = await tx.salesOrder.findUnique({
      where: { id: orderId },
      include: { items: true, feeItems: true },
    });

    // 1. 计算费用总额
    const totalExpense = order.feeItems.reduce((sum, fee) => sum + fee.feeAmount, 0);

    // 2. 按金额比例分摊费用
    const itemsForAllocation = order.items.map(item => ({
      id: item.id,
      totalPrice: item.subtotal,
    }));

    const expenseAllocations = allocateExpensesByValue(
      itemsForAllocation,
      totalExpense
    );

    // 3. 对每个订单项创建出库记录
    for (const item of order.items) {
      const inventory = await tx.inventory.findFirst({
        where: { productId: item.productId, quantity: { gt: 0 } },
      });

      // 3.1 计算单位成本（库存成本 + 分摊费用）
      const inventoryUnitCost = inventory?.unitCost || 0;
      const allocatedExpense = expenseAllocations.get(item.id) || 0;
      const unitCost = inventoryUnitCost + allocatedExpense / item.quantity;
      const totalCost = unitCost * item.quantity;

      // 3.2 更新订单项成本
      await tx.salesOrderItem.update({
        where: { id: item.id },
        data: {
          unitCost,
          costSubtotal: totalCost,
          allocatedExpense,
          profitAmount: item.subtotal - totalCost,
        },
      });

      // 3.3 创建出库记录
      await tx.outboundRecord.create({
        data: {
          productId: item.productId,
          quantity: item.quantity,
          unitCost,
          totalCost,
          salesOrderId: orderId,
          // ...
        },
      });
    }

    // 4. 更新订单总成本和利润
    const totalCost = order.items.reduce((sum, item) => {
      const allocatedExpense = expenseAllocations.get(item.id) || 0;
      const inventoryUnitCost = /* 获取库存成本 */;
      return sum + (inventoryUnitCost + allocatedExpense / item.quantity) * item.quantity;
    }, 0);

    await tx.salesOrder.update({
      where: { id: orderId },
      data: {
        costAmount: totalCost,
        expenseAmount: totalExpense,
        profitAmount: order.totalAmount - totalCost - totalExpense,
        status: 'shipped',
      },
    });
  });
}
```

**步骤2: 创建Excel导入API**

```typescript
// app/api/inventory/initial-stock/import/route.ts

import { NextRequest, NextResponse } from 'next/server';
import * as XLSX from 'xlsx';
import { z } from 'zod';

// 导入数据验证Schema
const initialStockRowSchema = z.object({
  productCode: z.string().min(1, '产品编码不能为空'),
  productName: z.string().min(1, '产品名称不能为空'),
  colorCode: z.string().optional(),
  batchNumber: z.string().optional(),
  quantity: z.number().positive('数量必须大于0'),
  unitCost: z.number().nonnegative('单位成本不能为负'),
  totalCost: z.number().nonnegative('总成本不能为负'),
  costSource: z.enum(['财务查账', '供应商询价', '市场估值']),
  location: z.string().optional(),
  remarks: z.string().optional(),
});

export async function POST(request: NextRequest) {
  try {
    // 1. 解析Excel文件
    const formData = await request.formData();
    const file = formData.get('file') as File;

    const buffer = await file.arrayBuffer();
    const workbook = XLSX.read(buffer);
    const worksheet = workbook.Sheets[workbook.SheetNames[0]];
    const rawData = XLSX.utils.sheet_to_json(worksheet);

    // 2. 数据验证
    const validatedRows = [];
    const errors = [];

    for (let i = 0; i < rawData.length; i++) {
      try {
        const row = rawData[i];
        const validated = initialStockRowSchema.parse({
          productCode: row['产品编码'],
          productName: row['产品名称'],
          colorCode: row['色号'],
          batchNumber: row['批次号'],
          quantity: Number(row['数量']),
          unitCost: Number(row['单位成本']),
          totalCost: Number(row['总成本']),
          costSource: row['成本来源'],
          location: row['存储位置'],
          remarks: row['备注'],
        });

        // 验证总成本 = 数量 × 单位成本
        if (Math.abs(validated.totalCost - validated.quantity * validated.unitCost) > 0.01) {
          errors.push({
            row: i + 2,
            error: '总成本与数量×单位成本不匹配',
          });
          continue;
        }

        validatedRows.push(validated);
      } catch (error) {
        errors.push({
          row: i + 2,
          error: error.message,
        });
      }
    }

    // 3. 如果有错误，返回错误信息
    if (errors.length > 0) {
      return NextResponse.json({
        success: false,
        errors,
      }, { status: 400 });
    }

    // 4. 批量导入
    const results = await prisma.$transaction(async tx => {
      const importResults = [];

      for (const row of validatedRows) {
        // 4.1 查找产品
        const product = await tx.product.findFirst({
          where: { code: row.productCode },
        });

        if (!product) {
          importResults.push({
            productCode: row.productCode,
            success: false,
            error: '产品不存在',
          });
          continue;
        }

        // 4.2 查找或创建产品变体（如果有色号）
        let variantId = null;
        if (row.colorCode) {
          let variant = await tx.productVariant.findFirst({
            where: {
              productId: product.id,
              colorCode: row.colorCode,
            },
          });

          if (!variant) {
            variant = await tx.productVariant.create({
              data: {
                productId: product.id,
                colorCode: row.colorCode,
                colorName: row.colorCode,
                sku: `${product.code}-${row.colorCode}`,
              },
            });
          }

          variantId = variant.id;
        }

        // 4.3 创建期初入库记录
        const inboundRecord = await tx.inboundRecord.create({
          data: {
            recordNumber: `INIT-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
            productId: product.id,
            variantId,
            batchNumber: row.batchNumber,
            quantity: row.quantity,
            unitCost: row.unitCost,
            totalCost: row.totalCost,
            reason: 'initial_stock',
            remarks: `${row.costSource} - ${row.remarks || ''}`,
            userId: /* 当前用户ID */,
          },
        });

        // 4.4 更新库存
        const existingInventory = await tx.inventory.findFirst({
          where: {
            productId: product.id,
            variantId,
            batchNumber: row.batchNumber,
          },
        });

        if (existingInventory) {
          await tx.inventory.update({
            where: { id: existingInventory.id },
            data: {
              quantity: { increment: row.quantity },
              unitCost: row.unitCost,
              location: row.location,
            },
          });
        } else {
          await tx.inventory.create({
            data: {
              productId: product.id,
              variantId,
              batchNumber: row.batchNumber,
              quantity: row.quantity,
              unitCost: row.unitCost,
              location: row.location,
            },
          });
        }

        importResults.push({
          productCode: row.productCode,
          success: true,
          inboundRecordId: inboundRecord.id,
        });
      }

      return importResults;
    });

    return NextResponse.json({
      success: true,
      data: {
        total: validatedRows.length,
        succeeded: results.filter(r => r.success).length,
        failed: results.filter(r => !r.success).length,
        results,
      },
    });
  } catch (error) {
    return NextResponse.json({
      success: false,
      error: error.message,
    }, { status: 500 });
  }
}
```
