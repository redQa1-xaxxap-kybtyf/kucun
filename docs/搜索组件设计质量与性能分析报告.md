#### 2.1 防抖机制实现优秀

<augment_code_snippet path="hooks/use-debounced-search.ts" mode="EXCERPT">

```typescript
export function useDebouncedSearch(options = {}) {
  const { delay = 400, minLength = 0, cancelOnUnmount = true } = options;

  // ✅ 使用 useRef 避免闭包陷阱
  const timeoutRef = useRef<NodeJS.Timeout | null>(null);
  const mountedRef = useRef(true);

  // ✅ 智能防抖逻辑
  useEffect(() => {
    if (inputValue.length < minLength) {
      setDebouncedValue(''); // 低于最小长度直接清空
      return;
    }

    if (inputValue === debouncedValue) {
      return; // 值未变化,跳过防抖
    }

    setIsDebouncing(true);
    timeoutRef.current = setTimeout(() => {
      if (mountedRef.current) {
        // ✅ 防止内存泄漏
        setDebouncedValue(inputValue);
      }
    }, delay);

    return () => clearTimeout(timeoutRef.current); // ✅ 清理定时器
  }, [inputValue, debouncedValue, delay, minLength]);
}
```

</augment_code_snippet>

**性能优化亮点**:

1. ✅ **智能跳过**: 值未变化时不触发防抖
2. ✅ **内存安全**: 组件卸载时清理定时器,防止内存泄漏
3. ✅ **可配置**: 支持自定义延迟、最小长度等参数
4. ✅ **状态指示**: 提供 `isDebouncing` 状态,可显示加载动画

**与 React Hook Form 最佳实践对比**:


### 3. **TanStack Query 集成优秀** ⭐⭐⭐⭐⭐

#### 3.1 GlobalSearch 的查询优化

<augment_code_snippet path="components/common/GlobalSearch.tsx" mode="EXCERPT">

```typescript
const { data: results = [], isLoading } = useQuery({
  queryKey: ['global-search', query.trim()],
  queryFn: async ({ signal }) => {
    if (!query.trim()) return [];

    // ✅ 并行搜索多个数据源
    const [productResults, orderResults, customerResults] = await Promise.all([
      searchProducts(query.trim(), signal),
      searchOrders(query.trim(), signal),
      searchCustomers(query.trim(), signal),
    ]);

    return [...productResults, ...orderResults, ...customerResults];
  },
  enabled: query.trim().length > 0, // ✅ 条件查询
  staleTime: 30 * 1000, // ✅ 30秒缓存
  gcTime: 5 * 60 * 1000, // ✅ 5分钟垃圾回收
});
```

</augment_code_snippet>

**符合 TanStack Query v5 最佳实践**:

1. ✅ **并行请求**: 使用 `Promise.all` 并行搜索,减少总耗时
2. ✅ **AbortController**: 支持请求取消,避免竞态条件
3. ✅ **条件查询**: `enabled` 避免空查询
4. ✅ **缓存策略**: `staleTime` 和 `gcTime` 配置合理
5. ✅ **Query Key 规范**: 使用 `['global-search', query]` 格式

**性能对比**:

### 问题 2: **缺少 Zod 验证** ⚠️

**当前状态**:

```typescript
// ❌ 没有使用 Zod 验证搜索参数
const { data: results } = useQuery({
  queryKey: ['global-search', query.trim()],
  queryFn: async () => {
    // 直接使用未验证的 query
    const response = await fetch(`/api/products/search?search=${query}`);
    return response.json(); // ❌ 返回值未验证
  },
});
```

**改进方案**: 添加 Zod Schema

```typescript
// ✅ 定义搜索参数 Schema
import { z } from 'zod';

const SearchParamsSchema = z.object({
  query: z.string().min(1).max(100),
  limit: z.number().int().min(1).max(50).default(5),
});

const SearchResultSchema = z.object({
  id: z.string(),
  title: z.string(),
  description: z.string().optional(),
  type: z.enum(['product', 'order', 'customer', 'document']),
  href: z.string().url(),
});

// ✅ 在 API 调用中使用
const searchProducts = async (query: string, signal?: AbortSignal) => {
  const params = SearchParamsSchema.parse({ query, limit: 5 });

  const response = await fetch(
    `/api/products/search?${new URLSearchParams(params)}`,
    { signal }
  );

  const data = await response.json();
  return z.array(SearchResultSchema).parse(data); // ✅ 验证返回值
};
```

**预期效果**:

- ✅ 运行时类型安全
- ✅ 自动过滤非法输入
- ✅ 更好的错误提示

**改进方案**: 添加错误处理

```typescript
// ✅ 使用 TanStack Query 的错误处理
const { data, isLoading, error } = useQuery({
  queryKey: ['global-search', query],
  queryFn: async ({ signal }) => {
    const response = await fetch(`/api/search?q=${query}`, { signal });

    if (!response.ok) {
      throw new Error(`搜索失败: ${response.statusText}`);
    }

    return response.json();
  },
  retry: 1, // 只重试一次
  retryDelay: 1000,
});

// ✅ 在 UI 中显示错误
{error && (
  <div className="text-destructive p-4 text-center">
    <AlertCircle className="mx-auto h-8 w-8" />
```
